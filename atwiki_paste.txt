  function normLine(line) {
    var text = String(line || '').replace(/\.{2,}/g, '……').replace(/…+/g, '……').replace(/・{3,}/g, '……').replace(/(?:--+|－{2,}|—+|―+)/g, '――').replace(/!/g, '！').replace(/\?/g, '？');
    var result = '', i = 0;
    while (i < text.length) {
      var ch = text.charAt(i);
      if (ch !== '！' && ch !== '？') { result += ch; i++; continue; }
      var marks = '';
      while (i < text.length && (text.charAt(i) === '！' || text.charAt(i) === '？')) marks += text.charAt(i++);
      result += marks;
      var rest = text.slice(i);
      if (!rest) continue;
      if (/^[ \t\u3000]/.test(rest)) { result += ' '; text = text.slice(0, i) + rest.replace(/^[ \t\u3000]+/, ''); continue; }
      if (/^[」』）】〉》〕］｝”’]+$/.test(rest)) continue;
      result += ' ';
    }
    return result;
  }

  function normalizeRelatedLink(value) {
    var text = String(value == null ? '' : value).trim();
    if (!text) return '';
    if (/^\[\[.*\]\]$/.test(text)) text = text.slice(2, -2).trim();
    text = text.replace(/\|/g, '&#124;');

  function relatedValuesForTalk(data, talkIndex) {
    var entries = [], regex = new RegExp('^relatedTalk' + talkIndex + '_(\\d+)$');
    Object.keys(data || {}).forEach(function (key) {
      var match = key.match(regex);
      if (!match) return;
      entries.push({ index: Number(match[1]), value: data[key] });
    });
    entries.sort(function (a, b) { return a.index - b.index; });
    return entries.map(function (entry) { return normalizeRelatedLink(entry.value); }).filter(Boolean);
  }

  function pushRows(lines, rows, data) {

    rows.forEach(function (row) {

      if (row.family === 'talk') {
        var match = row.key.match(/^talk(\d+)$/);
        if (match) value += relatedSuffixForTalk(data, Number(match[1]));
      }
      if (row.span) { lines.push('|>|' + row.label + '|' + value + '|'); return; }
      var first = seen[row.family] ? '~' : row.label;
      seen[row.family] = true;
      lines.push('|' + first + '|' + row.sub + '|' + value + '|');
    });
  }

  function normalizeStoredData(data) {

    if (filled(normalized.costume) && !filled(normalized.costumeTalk1)) normalized.costumeTalk1 = normalized.costume;
    delete normalized.costume;

      if (!match) return;
      var newKey = 'relatedTalk' + match[1] + '_1';
      if (!filled(normalized[newKey]) && filled(normalized[key])) normalized[newKey] = normalized[key];
      delete normalized[key];
    });
    return normalized;
  }

    }
  }

  function migrateV3(data) {
    if (!data || typeof data !== 'object') return null;
    var result = { normal: normalizeStoredData(data.normal || {}), normalCounts: data.normalCounts || {}, diffs: [] };
    if (!result.normalCounts.related) result.normalCounts.related = {};
    (data.diffs || []).forEach(function (item) {
      var counts = item.counts || {};
      if (!counts.related) counts.related = {};
      result.diffs.push({ title: item.title, data: normalizeStoredData(item.data || {}), counts: counts });
    });
    return result;
  }

  function migrateV2(data) {
    if (!data || typeof data !== 'object') return null;
    var normal = normalizeStoredData(data.normal || {});
    var result = { normal: normal, normalCounts: { talk: Number(data.normalTalkCount) || 3, costumeUnlock: 1, costumeTalk: 1, related: {} }, diffs: [] };
    (data.diffs || []).forEach(function (item) {
      result.diffs.push({ title: item.title, data: normalizeStoredData(item.data || {}), counts: { talk: Number(item.talkCount) || 3, costumeUnlock: 1, costumeTalk: 1, related: {} } });
    });
    return result;
  }

  function migrateLegacy(data) {
    if (!data || typeof data !== 'object') return null;
    var normal = {}, diff = {};
    Object.keys(data).forEach(function (key) {
      if (key.indexOf('normal_') === 0) normal[key.slice(7)] = data[key];
      else if (key.indexOf('trueName_') === 0) diff[key.slice(9)] = data[key];
    });
    normal = normalizeStoredData(normal);
    diff = normalizeStoredData(diff);
    var result = { normal: normal, normalCounts: { talk: 3, costumeUnlock: 1, costumeTalk: 1, related: {} }, diffs: [] };
    if (Object.keys(diff).some(function (key) { return filled(diff[key]); })) result.diffs.push({ title: '真名判明時', data: diff, counts: { talk: 3, costumeUnlock: 1, costumeTalk: 1, related: {} } });
    return result;
  }

  function restore() {
    try {
      var raw = localStorage.getItem(STORAGE_KEY);
      var data = raw ? JSON.parse(raw) : null;
      if (!data) {
        var previousRaw = localStorage.getItem(PREVIOUS_KEY);
        if (previousRaw) data = migrateV3(JSON.parse(previousRaw));
      }
      if (!data) {
        var v2Raw = localStorage.getItem(V2_KEY);
        if (v2Raw) data = migrateV2(JSON.parse(v2Raw));
      }
      if (!data) {
        var legacyRaw = localStorage.getItem(LEGACY_KEY);
        if (legacyRaw) data = migrateLegacy(JSON.parse(legacyRaw));
      }
      if (!data) return;
      var normalEditor = normalHost.querySelector('.ra-editor');
      prepareEditor(normalEditor, data.normal || {}, data.normalCounts || {});
      (data.diffs || []).forEach(function (item) { addDiff(item.title, item.data || {}, item.counts || {}); });
      setStatus('前回保存した入力内容を復元しました。');
    } catch (e) {
      setStatus('保存データの復元に失敗しました。', true);
    }
  }

  function generate() { outputEl.value = output(); save(false); setStatus('生成しました。未入力行と空セクションは省略しています。'); }
  function fallbackCopy() { outputEl.focus(); outputEl.select(); try { document.execCommand('copy'); setStatus('生成結果をクリップボードへコピーしました。'); } catch (e) { setStatus('自動コピーに失敗しました。生成結果を選択して手動でコピーしてください。', true); } }
  function copy() { if (!outputEl.value.trim()) generate(); if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(outputEl.value).then(function () { setStatus('生成結果をクリップボードへコピーしました。'); }).catch(fallbackCopy); else fallbackCopy(); }

  function resetDynamicRows(editor, config) {
    var list = editor.querySelector('[' + config.listAttr + ']');
    if (!list) return;
    Array.prototype.slice.call(list.querySelectorAll('[' + config.indexAttr + ']')).forEach(function (row) {
      var index = Number(row.getAttribute(config.indexAttr));
      if (index > config.initial && row.parentNode) row.parentNode.removeChild(row);
    });
  }
  function resetRelatedInputs(editor) {
    Array.prototype.forEach.call(editor.querySelectorAll('.ra-talk-row'), function (talkRowEl) {
      var list = talkRowEl.querySelector('.ra-related-list');
      Array.prototype.slice.call(list.querySelectorAll('[data-related-index]')).forEach(function (row) {
        if (Number(row.getAttribute('data-related-index')) > 1 && row.parentNode) row.parentNode.removeChild(row);
      });
    });
  }

  function clearAll() {
    if (!window.confirm('入力内容と保存データをすべて消去します。よろしいですか？')) return;
    localStorage.removeItem(STORAGE_KEY);
    localStorage.removeItem(PREVIOUS_KEY);
    localStorage.removeItem(V2_KEY);
    localStorage.removeItem(LEGACY_KEY);
    var normalEditor = normalHost.querySelector('.ra-editor');
    resetDynamicRows(normalEditor, DYNAMIC_CONFIGS.talk);
    resetDynamicRows(normalEditor, DYNAMIC_CONFIGS.costumeUnlock);
    resetDynamicRows(normalEditor, DYNAMIC_CONFIGS.costumeTalk);
    resetRelatedInputs(normalEditor);
    Array.prototype.forEach.call(normalEditor.querySelectorAll('textarea,input[type=text]'), function (el) { el.value = ''; });
    diffsHost.innerHTML = '';
    diffSerial = 0;
    outputEl.value = '';
    setStatus('クリアしました。');
  }

  root.querySelector('#ra-copy').addEventListener('click', copy);
  root.querySelector('#ra-save').addEventListener('click', function () { save(true); });
  root.querySelector('#ra-clear').addEventListener('click', clearAll);

    if (event.target.closest('[data-add-costume-talk]')) { appendDynamicRow(editor, DYNAMIC_CONFIGS.costumeTalk); return; }
    var removeDiffButton = event.target.closest('[data-remove-diff]');
    if (removeDiffButton) {
      var diffEditor = removeDiffButton.closest('.ra-diff');
      if (diffEditor && diffEditor.parentNode) diffEditor.parentNode.removeChild(diffEditor);
    }
  });
